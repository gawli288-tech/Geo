const generateMockHistory = (baseLat: number, baseLng: number): any[] => {
  const history = [];
  const now = new Date();
  for (let i = 5; i > 0; i--) {
    const time = new Date(now.getTime() - i * 30 * 60000); // Every 30 mins
    history.push({
      id: `h_${i}`,
      timestamp: time,
      location: {
        lat: baseLat + (Math.random() - 0.5) * 0.01,
        lng: baseLng + (Math.random() - 0.5) * 0.01
      },
      label: i === 5 ? 'Home' : (i === 3 ? 'Coffee Shop' : undefined)
    });
  }
  return history;
};
